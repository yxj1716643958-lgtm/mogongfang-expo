# 微信小程序支付集成指南

## 整体架构

```
用户浏览器 → H5页面 → 点击微信支付
    ↓
生成小程序URL Scheme
    ↓
跳转微信小程序
    ↓
小程序调起汇付支付 (trad_type=T_MINIAPP)
    ↓
支付完成 → 返回H5页面
```

## 第一步：创建微信小程序

### 1.1 注册小程序
1. 访问 https://mp.weixin.qq.com/
2. 注册小程序账号（需要企业资质）
3. 完成认证（300元/年）

### 1.2 配置小程序
1. 登录小程序后台
2. 获取 **AppID** 和 **AppSecret**
3. 开通微信支付功能

## 第二步：配置服务器

### 2.2 获取用户 OpenID

在小程序中获取用户 OpenID：

```javascript
// 小程序代码
wx.login({
  success: (res) => {
    if (res.code) {
      // 发送 code 到服务器换取 OpenID
      wx.request({
        url: 'https://your-server.com/api/v1/wechat/code2session',
        data: { code: res.code },
        success: (response) => {
          const { openid } = response.data;
          // 保存 OpenID 用于支付
        }
      });
    }
  }
});
```

服务器端实现 code2session：

```javascript
// ticket-server.js
app.get('/api/v1/wechat/code2session', async (req, res) => {
  const { code } = req.query;

  const response = await axios.get(
    `https://api.weixin.qq.com/sns/jscode2session?appid=${WECHAT_MINI_APPID}&secret=${WECHAT_MINI_APPSECRET}&js_code=${code}&grant_type=authorization_code`
  );

  res.json({
    openid: response.data.openid,
    session_key: response.data.session_key
  });
});
```

### 3.2 小程序支付逻辑 (payment.js)

```javascript
Page({
  data: {
    orderNo: '',
    amount: 0,
    paying: false
  },

  onLoad(options) {
    const { orderNo, amount } = options;
    this.setData({ orderNo, amount });
  },

  async startPayment() {
    this.setData({ paying: true });

    try {
      // 1. 调用后端创建支付
      const payResult = await this.requestPayment({
        orderNo: this.data.orderNo,
        paymentMethod: 'WECHAT_MINIAPP'
      });

      // 2. 调起微信支付
      wx.requestPayment({
        timeStamp: payResult.timeStamp,
        nonceStr: payResult.nonceStr,
        package: payResult.package,
        signType: 'MD5',
        paySign: payResult.paySign,
        success: () => {
          wx.showToast({ title: '支付成功' });
          setTimeout(() => {
            wx.navigateBack();
          }, 1500);
        },
        fail: (error) => {
          wx.showToast({
            title: '支付失败',
            icon: 'none'
          });
        }
      });

    } catch (error) {
      wx.showToast({
        title: error.message || '支付失败',
        icon: 'none'
      });
    } finally {
      this.setData({ paying: false });
    }
  },

  requestPayment(params) {
    return new Promise((resolve, reject) => {
      wx.request({
        url: 'https://your-server.com/api/v1/payments/miniapp/create',
        data: params,
        method: 'POST',
        success: (res) => {
          if (res.data.code === 'SUCCESS') {
            resolve(res.data.data);
          } else {
            reject(new Error(res.data.message));
          }
        },
        fail: reject
      });
    });
  }
});
```

## 第四步：H5页面跳转小程序

### 4.1 更新 tickets.html

```javascript
// 支付方式选择
async function selectPaymentMethod(method) {
  if (method === 'wechat') {
    // 微信支付：跳转小程序
    await launchMiniProgram(orderNo);
  } else if (method === 'alipay') {
    // 支付宝支付：显示二维码
    await showAlipayQR(orderNo);
  }
}

// 跳转微信小程序
async function launchMiniProgram(orderNo) {
  try {
    // 1. 生成小程序 URL Scheme
    const response = await fetch('/api/v1/wechat/miniprogram/scheme', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        orderNo,
        path: 'pages/payment/payment',
        query: `orderNo=${orderNo}`
      })
    });

    const { openlink } = await response.json();

    // 2. 跳转到小程序
    window.location.href = openlink;

  } catch (error) {
    console.error('跳转小程序失败:', error);
    alert('请使用微信扫码或选择支付宝支付');
  }
}
```

### 4.2 支付宝支付显示二维码

```javascript
async function showAlipayQR(orderNo) {
  try {
    const response = await fetch('/api/v1/payments/alipay/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ orderNo })
    });

    const { code, data } = await response.json();

    if (code === 'SUCCESS') {
      // 显示支付宝支付二维码
      document.getElementById('alipayQR').src = data.qrCode;
      document.getElementById('paymentModal').classList.remove('hidden');
    }
  } catch (error) {
    console.error('创建支付失败:', error);
  }
}
```

## 第五步：后端实现

### 5.1 创建小程序 URL Scheme

```javascript
// ticket-server.js
app.post('/api/v1/wechat/miniprogram/scheme', async (req, res) => {
  const { orderNo, path, query } = req.body;

  try {
    // 调用微信API生成 URL Scheme
    const response = await axios.post(
      `https://api.weixin.qq.com/wxa/generatescheme?access_token=${await getAccessToken()}`,
      {
        jump_wxa: {
          path: path,
          query: query
        },
        is_expire: true,
        expire_type: 1,
        expire_interval: 30  // 30天后过期
      }
    );

    res.json({
      code: 'SUCCESS',
      data: {
        openlink: response.data.openlink
      }
    });

  } catch (error) {
    res.json({
      code: 'FAIL',
      message: '生成小程序跳转链接失败'
    });
  }
});

// 获取小程序 Access Token
async function getAccessToken() {
  const response = await axios.get(
    `https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid=${WECHAT_MINI_APPID}&secret=${WECHAT_MINI_APPSECRET}`
  );
  return response.data.access_token;
}
```

### 5.2 创建小程序支付

```javascript
app.post('/api/v1/payments/miniapp/create', async (req, res) => {
  const { orderNo, paymentMethod } = req.body;

  try {
    // 1. 查询订单
    const [orders] = await pool.query(
      'SELECT * FROM orders WHERE order_no = ?',
      [orderNo]
    );

    if (orders.length === 0) {
      throw new Error('订单不存在');
    }

    const order = orders[0];

    // 2. 构建汇付支付请求
    const paymentRequest = HuifuRequestBuilder.buildMiniAppPayment({
      orderNo,
      transAmt: Math.round(order.total_amount * 100),
      goodsDesc: '门票',
      wxAppid: WECHAT_MINI_APPID,
      wxOpenid: req.body.wxOpenid  // 从小程序传递的 OpenID
    });

    // 3. 调用汇付API
    const huifuResult = await HuifuHttpClient.request(
      '/gateway/api/rest/api/transaction/acqTrade',
      paymentRequest
    );

    // 4. 返回支付参数
    res.json({
      code: 'SUCCESS',
      data: {
        timeStamp: Math.floor(Date.now() / 1000).toString(),
        nonceStr: generateNonceStr(),
        package: `prepay_id=${huifuResult.prepay_id}`,
        signType: 'MD5',
        paySign: huifuResult.sign
      }
    });

  } catch (error) {
    res.json({
      code: 'FAIL',
      message: error.message
    });
  }
});
```
